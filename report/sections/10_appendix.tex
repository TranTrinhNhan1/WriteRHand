\section{Tái lập kết quả}
\label{app:repro}
Môi trường dùng \texttt{uv} (Python $\ge3.11$). Mỗi lệnh nhận thêm \texttt{--dataset fashion\_mnist} hoặc \texttt{--dataset emnist\_digits}. Notebook \texttt{notebooks/reproduce\_results.ipynb} chạy lại các bước chính trên tập con nhỏ.
{\scriptsize\begin{verbatim}
uv sync --extra track --extra export
uv run python -m src prepare            # tải dữ liệu, chia train/val, chuẩn hóa
uv run python -m src tune --model all   # quét siêu tham số -> results/tuning
uv run python -m src train --model all  # huấn luyện cuối, đánh giá 3 tập
uv run python -m src baselines          # scikit-learn, cùng siêu tham số
uv run python -m src augment --model mlp; uv run python -m src augment --model cnn --seeds 2
uv run python -m src compare            # CNN so với MLP, kiểm tra chéo
uv run python -m src report             # bảng, hình, đoạn LaTeX
uv run python -m src export --site      # ONNX + kiểm tra số học
\end{verbatim}}
